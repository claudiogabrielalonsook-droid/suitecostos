WITH Ultimos AS (
    SELECT
        *,
        ROW_NUMBER() OVER (
            PARTITION BY ObjID_TBCustoMat
            ORDER BY DataRef DESC, ObjID DESC
        ) AS rn
    FROM dbo.TBCustoMatItem
)
SELECT 
    u.ObjID,
    u.ObjID_TBCustoMat,
    u.DataRef,
    u.PrecoRevenda,
    u.IdMoeda,
    t.CodTabela,
    t.Descricao,
    t.UnidadeCusto
FROM Ultimos u
JOIN dbo.TBCustoMat t ON u.ObjID_TBCustoMat = t.ObjID
WHERE u.rn = 1
ORDER BY u.DataRef DESC
